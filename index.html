function sendTomorrowShift() {
  const YEAR = 2026, MONTH = 11;                    // 근무표의 연·월
  const sh = SpreadsheetApp.getActive().getSheetByName('11월 연습용'); // 탭 이름이 다르면 수정
  const t = new Date();
  t.setDate(t.getDate() + 1);
  if (t.getFullYear() !== YEAR || t.getMonth() + 1 !== MONTH) return;
  const label = { D: '데이', E: '이브닝', N: '나이트' };
  const rows = sh.getRange(4, 1, 15, 40).getValues(); // A4:AN18
  rows.forEach(r => {
    const shift = r[t.getDate() + 2]; // 날짜 d의 근무는 D열부터 (인덱스 d+2)
    const mail = r[39];               // AN열 이메일
    if (label[shift] && mail) {
      MailApp.sendEmail(mail, '[내일 근무 안내] ' + label[shift],
        r[0] + '번 간호사님, 내일(' + (t.getMonth() + 1) + '/' + t.getDate() + ') 근무는 ' + label[shift] + '입니다.');
    }
  });
}
